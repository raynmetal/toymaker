470 PhysicsSystem::ConstraintID PhysicsSystem::registerConstraint(
471 const typename TConstraint::Config& config,
472 const std::vector<std::pair<EntityID, typename TConstraint::Parameter>>& participants,
473 float compliance
474 ) {
475 // reclaim a deleted constraint id if possible
476 ConstraintID newConstraint { mConstraints.size() };
477 if(!mConstraintsDeleted.empty()) {
478 newConstraint = *mConstraintsDeleted.begin();
479 mConstraintsDeleted.erase(newConstraint);
480 }
481
482 // get separate constraint parameter and participating entity lists
483 std::unordered_set<EntityID> registeredEntities {};
484 std::vector<EntityID> entities {};
485 std::vector<typename TConstraint::Parameter> constraintData {};
486 for(const std::pair<EntityID, typename TConstraint::Parameter>& participant: participants) {
487 const bool alreadyRegistered {
488 registeredEntities.find(participant.first) != registeredEntities.end()
489 };
490 assert(!alreadyRegistered && "An entity has appeared twice in the constraint registration list, which is invalid.");
491 constraintData.push_back(participant.second);
492 entities.push_back(participant.first);
493 registeredEntities.insert(participant.first);
494 }
495
496 // create constraint
497 if(newConstraint == mConstraints.size()) {
498 mConstraints.push_back({
499 std::make_unique<TConstraint>(config, constraintData, compliance),
500 entities
501 });
502 } else {
503 mConstraints[newConstraint] = {
504 std::make_unique<TConstraint>(config, constraintData, compliance),
505 entities
506 };
507 }
508
509 for(const auto& entity: entities) {
510 mEntityConstraintMap[entity].insert(newConstraint);
511 }
512
513 return newConstraint;
514 }
